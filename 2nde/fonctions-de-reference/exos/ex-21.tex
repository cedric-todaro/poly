\begin{exo}{}
	Résoudre les inéquations suivantes dans $\R$ :
	\vspace*{-2mm}
	\begin{multicols}{2}
		\begin{enumerate}[itemsep=2mm]
			\item $\sqrt{x} \le 3$
			\item $x^3 = 2\sqrt{2}$
			\item $x^3 < -8$
			\item $\sqrt{x} = -1$
			\item $\sqrt{x} < \pi$
		\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
