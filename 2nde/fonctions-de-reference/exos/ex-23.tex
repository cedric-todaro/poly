\begin{exo}{}
	Résoudre les équations suivantes dans $\R$ :
	\vspace*{-2mm}
	\begin{multicols}{2}
		\begin{enumerate}[itemsep=2mm]
			\item $x^2 = 12$
			\item $x^2 = -\dfrac{5}{3}$
			\item $\sqrt{x+1} = 3$
			\item $\sqrt[3]{x} = -\dfrac{2}{5}$
			\item $\dfrac{3}{x} = \dfrac{6}{7}$
			\item $(x-3)^2 = 4$
		\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
