\begin{exo}{}
	Donner l'image par la fonction inverse $i : x \mapsto \dfrac{1}{x}$ des nombres suivants.
	\vspace*{-2mm}
	\begin{multicols}{2}
		\begin{enumerate}
			\item $2 + 3$
			\item $2 \times 3$
			\item $2 - 3$
			\item $2 \times (-3)$
			\item $\dfrac{2}{3}$
			\item $\dfrac{2}{-3}$
		\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
