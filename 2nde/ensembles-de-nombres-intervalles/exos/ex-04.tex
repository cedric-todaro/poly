		\item $\dfrac{3}{4} \ \dots \ \mathbb{Q}$
		\item $-\dfrac{6}{7} \ \dots \ \mathbb{R}$
		\item $\sqrt{4} \ \dots \ \mathbb{Z}$
		      % \item $-\sqrt{144} \ \dots \ \mathbb{Z}$
		      % \item $-17,2 \ \dots \ \mathbb{Q}$
		      % \item $\dfrac{10}{3} \ \dots \ \mathbb{D}$
	\end{enumerate}
	\end{multicols}
