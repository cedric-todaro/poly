\begin{exo}{}
	Écrire sous la forme $\sqrt{a}$ avec $a > 0$.
	\vspace*{-2mm}\begin{multicols}{2}
		\begin{enumerate}[itemsep=2mm]
			\item $\sqrt{7} \times \sqrt{6}$
			\item $\sqrt{15} \div \sqrt{5}$
			\item $\sqrt{16} + \sqrt{9}$
			\item $4\sqrt{3}$
		\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
