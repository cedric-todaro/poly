			\item $\dfrac{7}{12} \div \pa{\dfrac{5}{6} + \dfrac{1}{2}}$
			\item $\pa{\dfrac{3}{5}}^2 - \dfrac{-5}{6}$
		\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
