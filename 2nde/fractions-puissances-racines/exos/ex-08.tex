\begin{exo}{}
	Déterminer les valeurs exactes des nombres suivants.
	\vspace*{-2mm}\begin{multicols}{2}
		\begin{enumerate}[itemsep=2mm]
			\item $\sqrt{7^2 + 3^2}$
			\item $\sqrt{169 \times 144}$
			\item $\sqrt{169} - \sqrt{144}$
			\item $\sqrt{169 - 144}$
		\end{enumerate}
	\end{multicols}
\end{exo}
