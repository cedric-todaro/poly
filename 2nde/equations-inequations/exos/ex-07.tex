\begin{exo}[Résolution directe]
	Résoudre dans $\R$ les équations suivantes.

	On donnera les solutions sous forme exacte (racine carrée si nécessaire).
	\vspace*{-2mm}
	\begin{multicols}{2}
		\begin{enumerate}[label=\alph*),itemsep=2mm]
			\item $x^2 = 9$
			\item $x^2 = 5$
			\item $x^2 = -4$
			\item $3x^2 = 48$
			\item $2x^2 - 14 = 0$
			\item $\dfrac{-2}{5}x^2 + 2 = 0$
		\end{enumerate}
	\end{multicols}
\end{exo}
