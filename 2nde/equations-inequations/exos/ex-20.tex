\begin{exo}{}
	Résoudre les inéquations suivantes dans $\mathbb{R}$.
	\vspace*{-2mm}\begin{multicols}{2}
		\begin{enumerate}[itemsep=2mm]
			\item $\dfrac{x + 2}{-x + 6} < 0$
			\item $\dfrac{3x - 4}{2x + 3} \geqslant 0$
			\item $\dfrac{1 - 4x}{x - 3} < -4$
		\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
