\begin{exo}{}
	Dans chaque cas, déterminer les variations de la fonction affine $f$ définie sur $\R$.
	\vspace*{-2mm}
	\begin{multicols}{2}
	\begin{enumerate}
		\item $f(x) = 3x + 2$
		\item $f(x) = -6x$
		\item $f(x) = \dfrac{1}{6}$
		\item $f(x) = \dfrac{1}{6}x - \dfrac{1}{6}$
		\item $f(x) = 6 - \dfrac{1}{6}x$
		\item $f(x) = 6$
	\end{enumerate}
	\end{multicols}
\end{exo}
