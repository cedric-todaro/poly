\begin{exo}[Calculs arithmétiques]
	Soient \lstinline|x = 17| et \lstinline|y = 5|. Donner le résultat et le type des expressions suivantes :
	\begin{enumerate}[label=\alph*)]
		\item \lstinline|x / y|
		\item \lstinline|x // y|
		\item \lstinline|x % y|
		\item \lstinline|x ** 2|
	\end{enumerate}
\end{exo}
