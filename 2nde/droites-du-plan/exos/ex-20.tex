\begin{exo}{}
	Représenter dans le repère chacune des droites suivantes dont on donne une équation cartésienne :
	\vspace*{-2mm}
	\begin{multicols}{2}
		\begin{enumerate}
			\item $d_1 : x + y + 1 = 0$
			\item $d_2 : 2x - y - 2 = 0$
			\item $d_3 : -x + 2y + 3 = 0$
			\item $d_4 : 3x - 2y + 3 = 0$
			\item $d_5 : 2x + 3y - 4 = 0$
		\end{enumerate}
	\end{multicols}
\end{exo}
