\begin{exo}{}
	Dans chaque cas, déterminer si les droites $d$ et $d'$ sont strictement parallèles, sécantes ou confondues.
	\begin{enumerate}
		\item $d : 3x - 2y + 7 = 0$ et $d' : 3x + 2y - 7 = 0$
		\item $d : \dfrac{7}{3}x - y + 2 = 0$ et $d' : 7x - 3y + 9 = 0$
		\item $d : x + 7 = 0$ et $d' : -5x - 35 = 0$
	\end{enumerate}
\end{exo}
