\begin{exo}[Développement]
	Développer et réduire les expressions suivantes:
	\vspace*{-2mm}
	\begin{multicols}{2}
		\begin{enumerate}[label=\alph*)]
			\item $3(x + 2) - 4(2 - 2x)$
			\item $x(2x - 1) - 3(5 - x)$
			\item $(3x + 1)x - 3(x - 2)$
			\item $3(x - 5) - 2x(1 - 2x)$
		\end{enumerate}
	\end{multicols}
\end{exo}
