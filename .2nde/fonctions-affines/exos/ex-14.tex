\begin{exo}{}
	Classer les fonctions suivantes selon leur variation (croissante, décroissante, constante) sur $\R$.
	\begin{enumerate}
		\item $g(x) = 4x - 3$
		\item $h(x) = (-6x + 1) - 2(4x - 3)$
		\item $i(x) = 2(-6x + 1) + 3(4x - 3)$
		\item $j(x) = -3(-6x + 1) - (4x - 3)$
		\item $k(x) = (-6x + 1)^2 - 9x(4x - 3)$
	\end{enumerate}
\end{exo}
