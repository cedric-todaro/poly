\begin{exo}{}
	Établir le tableau de signes des fonctions affines suivantes sur $\R$.
	\vspace*{-2mm}
	\begin{multicols}{2}
	\begin{enumerate}
		\item $f(x) = 8x + 4$
		\item $g(x) = 8x$
		\item $h(x) = 4$
		\item $k(x) = -4x + 8$
		\item $\ell(x) = -4 + 8x$
	\end{enumerate}
	\end{multicols}
\end{exo}
