\begin{exo}{}
	Dans chaque cas, donner un encadrement de $\sqrt{x}$.
	\vspace*{-2mm}
	\begin{multicols}{2}
		\begin{enumerate}[itemsep=2mm]
			\item $1 < x < 2$
			\item $4 \le x < 12$
			\item $5 \le 4x < 16$
			\item $1,44 < x \le \pi^2 + 2\pi + 1$
		\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
