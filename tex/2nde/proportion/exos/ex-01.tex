\begin{exo}{}
	Sachant que, dans un village, il y a $500$ habitants et que $230$ d'entre eux sont des femmes, quelle est la proportion de femmes dans le village ?

	\vspace*{-2mm}\begin{multicols}{4}
	\begin{enumerate}
		\item $1,17$
		\item $0,54$
		\item $0,46$
		\item $2,17$
	\end{enumerate}
	\end{multicols}\vspace*{-4mm}
\end{exo}
