\begin{exo}{}
	Écrire sous la forme $a\sqrt{b}$ où $a$ et $b$ sont des nombres entiers strictement positifs, $b$ étant le plus petit possible.
	\vspace*{-2mm}\begin{multicols}{2}
		\begin{enumerate}[itemsep=2mm]
			\item $\sqrt{50}$
			\item $\sqrt{200}$
			\item $\sqrt{147}$
			\item $\sqrt{54}$
		\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
