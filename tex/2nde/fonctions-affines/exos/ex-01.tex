\begin{exo}{}
	Déterminer le coefficient directeur $m$ et l'ordonnée à l'origine $p$ de chaque fonction affine $f$ définie sur $\R$.
	\vspace*{-2mm}
	\begin{multicols}{2}
	\begin{enumerate}
		\item $f(x) = -2x + 3$
		\item $f(x) = -3 - 4x$
		\item $f(x) = -2$
		\item $f(x) = 3x$
		\item $f(x) = x + 4$
		\item $f(x) = 2 - x$
	\end{enumerate}
	\end{multicols}
\end{exo}
