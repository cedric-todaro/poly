\begin{exo}{}
	Dans chaque cas, déterminer le signe de $f(x)$ où $f$ est une fonction affine définie sur $\R$.
	\vspace*{-2mm}
	\begin{multicols}{2}
	\begin{enumerate}
		\item $f(x) = 3x + 1$
		\item $f(x) = 3$
		\item $f(x) = -x + 3$
		\item $f(x) = -3x$
		\item $f(x) = -3 + 3x$
		\item $f(x) = x + 3$
	\end{enumerate}
	\end{multicols}
\end{exo}
