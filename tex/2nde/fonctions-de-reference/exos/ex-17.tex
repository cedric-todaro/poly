\begin{exo}{}
	Résoudre les équations suivantes dans $\R$ :
	\vspace*{-2mm}
	\begin{multicols}{2}
		\begin{enumerate}[itemsep=2mm]
			\item $x^2 = 25$
			\item $x^2 = 5$
			\item $x^3 = -27$
			\item $\dfrac{1}{x} = -\dfrac{3}{8}$
			\item $\dfrac{1}{x} = 8$
			\item $\sqrt{x} = -3$
			\item $\sqrt{x} = 6$
			\item $x^3 = -1$
		\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
