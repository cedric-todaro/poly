\begin{exo}{}
	Sans calculatrice, ranger les nombres suivants par ordre croissant.
	\begin{enumerate}[itemsep=2mm]
		\item $(-3)^3 \ ; \ \pi^3 \ ; \ \sqrt{3}^3 \ ; \ (\sqrt{2} - 2)^3 \ ; \ 0$
		\item $\dfrac{-\pi^3}{8} \ ; \ 5\sqrt{5} \ ; \ 8 \ ; \ \dfrac{-27}{8} \ ; \ \dfrac{-64}{125} \ ; \ \sqrt{2}^3$
	\end{enumerate}
\end{exo}
