\begin{exo}{}
	Déterminer un encadrement de $x^3$ ou une inégalité que vérifie $x^3$ dans chacun des cas suivants.
	\vspace*{-2mm}
	\begin{multicols}{2}
		\begin{enumerate}[itemsep=2mm]
			\item $-3 \le x < 2$
			\item $-\sqrt{2} < 2x \le 1$
			\item $x \ge \dfrac{5}{6}$
		\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
