\begin{exo}{}
	Pour chaque cas, donner un encadrement de $x^2$, ou une inégalité vérifiée par $x^2$.
	\vspace*{-2mm}
	\begin{multicols}{2}
	\begin{enumerate}[itemsep=2mm]
		\item $-2 < x \le 7$
		\item $4 \le x < 7$
		\item $x > -3$
		\item $x < -2$
		\item $-6 \le x < 3$
		\item $-11 < x \le -2$
	\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
