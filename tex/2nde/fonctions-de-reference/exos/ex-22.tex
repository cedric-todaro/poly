\begin{exo}{}
	Résoudre les inéquations suivantes dans $\R$ :
	\vspace*{-2mm}
	\begin{multicols}{2}
	\begin{enumerate}[itemsep=2mm]
		\item $\dfrac{1}{x} = \dfrac{1}{7}$
		\item $\dfrac{1}{x} = \dfrac{5}{6}$
		\item $\dfrac{1}{x} \le -\dfrac{1}{8}$
		\item $\dfrac{1}{x} \le 12$
		\item $\dfrac{3}{x} \le 6$
	\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
