\begin{exo}{Avec la fonction carré}
	Sans calculatrice, compléter par $<$, $>$ ou $=$.
	\vspace*{-2mm}
	\begin{multicols}{2}
	\begin{enumerate}[itemsep=2mm]
		\item $3^2 \ \dots \ 4^2$
		\item $(-7)^2 \ \dots \ (-5)^2$
		\item $(-13,06)^2 \ \dots \ 13,06^2$
		\item $(-\pi)^2 \ \dots \ (-4)^2$
	\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
