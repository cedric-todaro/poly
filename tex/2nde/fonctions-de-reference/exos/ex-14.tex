\begin{exo}{}
	Déterminer un encadrement pour $\dfrac{1}{x}$ dans chacun des cas suivants.
	\vspace*{-2mm}
	\begin{multicols}{2}
	\begin{enumerate}[itemsep=2mm]
		\item $1 \le x \le 3$
		\item $\dfrac{1}{2} < x \le 6$
		\item $-5 < x < -1$
		\item $\dfrac{2\pi}{3} \le x < 12$
		\item $-1 + \dfrac{\pi}{4} < x \le -\dfrac{1}{16}$
	\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
