\begin{exo}{}
	Résoudre les inéquations suivantes dans $\R$ :
	\vspace*{-2mm}
	\begin{multicols}{2}
	\begin{enumerate}[itemsep=2mm]
		\item $x^2 \le 20$
		\item $2x^2 + 1 < 9$
		\item $-x^2 > -7$
		\item $x^2 + 1 \le \dfrac{1}{2}$
		\item $2x^2 - 3x \le x^2 - 3x + 8$
	\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
