\begin{exo}{}
	Donner l'inverse des nombres suivants.
	\vspace*{-2mm}
	\begin{multicols}{3}
	\begin{enumerate}
		\item $\dfrac{-1}{2} + \dfrac{1}{3}$
		\item $\dfrac{-1}{2} - \dfrac{1}{3}$
		\item $\dfrac{-1}{2} \times \dfrac{1}{3}$
		\item $\dfrac{-1}{2} \times \dfrac{-1}{3}$
		\item $\dfrac{\quad\dfrac{-1}{2}\quad}{\dfrac{1}{3}}$
		\item $\dfrac{\quad\dfrac{-1}{2}\quad}{\dfrac{-1}{3}}$
	\end{enumerate}
	\end{multicols}\vspace*{-2mm}
\end{exo}
