\begin{exo}[Développement]
	Développer et réduire les expressions suivantes:
	\begin{enumerate}[label=\alph*)]
		\item $(2x + 1)(3 - 2x)$
		\item $(x - 3)(-x - 1)$
		\item $(3x + 2)(5 - 2x)$
		\item $(x - 1)(3x^{2} - 2)$
		\item $(3 - x)(2x + 1) + 2(x + 2)$
		\item $(x - 1)(2x - 1) - 3(3 + 2x)$
	\end{enumerate}
\end{exo}
