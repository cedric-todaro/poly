\begin{exo}{}
	Résoudre les inéquations suivantes dans $\mathbb{R}$.
	\vspace*{-2mm}\begin{multicols}{2}
		\begin{enumerate}[itemsep=2mm]
			\item $7x + 3 > 2x - 5$
			\item $5x - 3 \leqslant 8x - 6$
			\item $7(x + 1) > 5 - 2x$
			\item $-5x + 3 \geqslant 2(x - 5)$
		\end{enumerate}
	\end{multicols}
\end{exo}
